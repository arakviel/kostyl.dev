```tsx
import { useState } from 'react';
import { View, Text, Pressable, StyleSheet, useColorScheme } from 'react-native';

export default function App() {
  const [count, setCount] = useState(0);
  const scheme = useColorScheme();
  const isDark = scheme === 'dark';

  return (
    <View style={styles.screen}>
      <Text style={[styles.label, { color: isDark ? '#f5f5f7' : '#111827' }]}>
        Натискань: {count}
      </Text>
      <Pressable
        onPress={() => setCount((c) => c + 1)}
        style={({ pressed }) => [styles.btn, pressed && styles.btnPressed]}
      >
        <Text style={styles.btnText}>Tap me</Text>
      </Pressable>
      <Text style={{ color: isDark ? '#a1a1aa' : '#64748b', fontSize: 12 }}>
        theme: {scheme}
      </Text>
    </View>
  );
}
```

```tsx [src/App.tsx]
import { useState } from 'react';
import { Counter } from './components/Counter';
import './styles.css';

export default function App() {
  const [theme, setTheme] = useState<'light' | 'dark'>('light');

  return (
    <div className={`app-shell ${theme === 'dark' ? 'dark-mode' : ''}`}>
      <header className="header">
        <h1>React Project Preview</h1>
        <button
          className="theme-btn"
          onClick={() => setTheme((t) => (t === 'light' ? 'dark' : 'light'))}
        >
          {theme === 'light' ? '🌙 Ніч' : '☀️ День'}
        </button>
      </header>

      <main className="content">
        <p className="desc">
          Багатофайловий приклад: дерево файлів зліва, живе прев'ю справа.
        </p>
        <Counter initial={5} step={2} />
      </main>
    </div>
  );
}
```

```tsx [src/components/Counter.tsx]
import { useState } from 'react';

interface CounterProps {
  initial?: number;
  step?: number;
}

export function Counter({ initial = 0, step = 1 }: CounterProps) {
  const [count, setCount] = useState(initial);

  return (
    <div className="counter-card">
      <div className="counter-display">
        <span className="counter-label">Поточне значення:</span>
        <span className="counter-number">{count}</span>
      </div>

      <div className="btn-group">
        <button className="btn btn-dec" onClick={() => setCount((c) => c - step)}>
          -{step}
        </button>
        <button className="btn btn-reset" onClick={() => setCount(0)}>
          Скинути
        </button>
        <button className="btn btn-inc" onClick={() => setCount((c) => c + step)}>
          +{step}
        </button>
      </div>
    </div>
  );
}
```

### Тест 4.2: Вкладені підпапки, JSON та Tailwind CSS v4 (Каталог курсів)

::react-project-preview{title="Каталог курсів IT" entry="src/App.tsx" :height="580"}

```tsx [src/App.tsx]
import { useState } from 'react';
import { CourseCard } from './components/CourseCard';
import coursesData from './data/courses.json';

export default function App() {
  const [filter, setFilter] = useState<'all' | 'react' | 'backend'>('all');

  const filtered = coursesData.filter((c) =>
    filter === 'all' ? true : c.tag === filter
  );

  return (
    <div className="p-6 max-w-4xl mx-auto font-sans">
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 mb-6 pb-4 border-b border-gray-200 dark:border-zinc-800">
        <div>
          <h1 className="text-2xl font-bold text-gray-900 dark:text-white">
            Навчальні курси
          </h1>
          <p className="text-sm text-gray-500 dark:text-gray-400">
            Демонстрація вкладених папок (ui/Badge), файлів JSON та Tailwind v4
          </p>
        </div>

        <div className="flex items-center gap-1.5 bg-gray-100 dark:bg-zinc-800/60 p-1 rounded-lg">
          <button
            onClick={() => setFilter('all')}
            className={`px-3 py-1 text-xs font-semibold rounded-md transition-all ${
              filter === 'all'
                ? 'bg-white dark:bg-zinc-700 text-sky-600 dark:text-sky-400 shadow-sm'
                : 'text-gray-600 dark:text-gray-400'
            }`}
          >
            Всі
          </button>
          <button
            onClick={() => setFilter('react')}
            className={`px-3 py-1 text-xs font-semibold rounded-md transition-all ${
              filter === 'react'
                ? 'bg-white dark:bg-zinc-700 text-sky-600 dark:text-sky-400 shadow-sm'
                : 'text-gray-600 dark:text-gray-400'
            }`}
          >
            React
          </button>
          <button
            onClick={() => setFilter('backend')}
            className={`px-3 py-1 text-xs font-semibold rounded-md transition-all ${
              filter === 'backend'
                ? 'bg-white dark:bg-zinc-700 text-sky-600 dark:text-sky-400 shadow-sm'
                : 'text-gray-600 dark:text-gray-400'
            }`}
          >
            Backend
          </button>
        </div>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
        {filtered.map((item) => (
          <CourseCard key={item.id} {...item} />
        ))}
      </div>
    </div>
  );
}
```

```tsx [src/components/ui/Badge.tsx]
interface BadgeProps {
  children: React.ReactNode;
  color?: 'blue' | 'purple' | 'amber';
}

export function Badge({ children, color = 'blue' }: BadgeProps) {
  const styles = {
    blue: 'bg-blue-50 text-blue-700 dark:bg-blue-950/50 dark:text-blue-300 border-blue-200 dark:border-blue-900',
    purple: 'bg-purple-50 text-purple-700 dark:bg-purple-950/50 dark:text-purple-300 border-purple-200 dark:border-purple-900',
    amber: 'bg-amber-50 text-amber-700 dark:bg-amber-950/50 dark:text-amber-300 border-amber-200 dark:border-amber-900',
  }[color];

  return (
    <span
      className={`inline-flex items-center px-2 py-0.5 rounded text-[11px] font-semibold border ${styles}`}
    >
      {children}
    </span>
  );
}
```
