### Тест 0.2: Pressable + state (Android)

::react-native-preview{title="Pressable counter" device="android" filename="Counter.tsx"}

```tsx
import { useState } from 'react';
import { View, Text, Pressable, StyleSheet, useColorScheme } from 'react-native';

export default function App() {
  const [count, setCount] = useState(0);
  const scheme = useColorScheme();
  const isDark = scheme === 'dark';

  return (
    <View style={styles.screen}>
      <Text style={[styles.label, { color: isDark ? '#f5f5f7' : '#111827' }]}>
        Натискань: {count}
      </Text>
      <Pressable
        onPress={() => setCount((c) => c + 1)}
        style={({ pressed }) => [styles.btn, pressed && styles.btnPressed]}
      >
        <Text style={styles.btnText}>Tap me</Text>
      </Pressable>
      <Text style={{ color: isDark ? '#a1a1aa' : '#64748b', fontSize: 12 }}>
        theme: {scheme}
      </Text>
    </View>
  );
}

const styles = StyleSheet.create({
  screen: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
    padding: 24,
    gap: 16,
  },
  label: {
    fontSize: 28,
    fontWeight: '700',
  },
  btn: {
    backgroundColor: '#2563EB',
    paddingVertical: 14,
    paddingHorizontal: 28,
    borderRadius: 14,
  },
  btnPressed: {
    opacity: 0.85,
    transform: [{ scale: 0.98 }],
  },
  btnText: {
    color: '#fff',
    fontSize: 16,
    fontWeight: '600',
  },
});
```

## 4. ReactProjectPreview — Multi-file React Sandbox

::react-project-preview{title="Інтерактивний лічильник з компонентами" entry="src/App.tsx" :height="540"}

```tsx [src/App.tsx]
import { useState } from 'react';
import { Counter } from './components/Counter';
import './styles.css';

export default function App() {
  const [theme, setTheme] = useState<'light' | 'dark'>('light');

  return (
    <div className={`app-shell ${theme === 'dark' ? 'dark-mode' : ''}`}>
      <header className="header">
        <h1>React Project Preview</h1>
        <button
          className="theme-btn"
          onClick={() => setTheme((t) => (t === 'light' ? 'dark' : 'light'))}
        >
          {theme === 'light' ? '🌙 Ніч' : '☀️ День'}
        </button>
      </header>

      <main className="content">
        <p className="desc">
          Багатофайловий приклад: дерево файлів зліва, живе прев'ю справа.
        </p>
        <Counter initial={5} step={2} />
      </main>
    </div>
  );
}
```

```tsx [src/components/Counter.tsx]
import { useState } from 'react';

interface CounterProps {
  initial?: number;
  step?: number;
}

export function Counter({ initial = 0, step = 1 }: CounterProps) {
  const [count, setCount] = useState(initial);

  return (
    <div className="counter-card">
      <div className="counter-display">
        <span className="counter-label">Поточне значення:</span>
        <span className="counter-number">{count}</span>
      </div>

      <div className="btn-group">
        <button className="btn btn-dec" onClick={() => setCount((c) => c - step)}>
          -{step}
        </button>
        <button className="btn btn-reset" onClick={() => setCount(0)}>
          Скинути
        </button>
        <button className="btn btn-inc" onClick={() => setCount((c) => c + step)}>
          +{step}
        </button>
      </div>
    </div>
  );
}
```
